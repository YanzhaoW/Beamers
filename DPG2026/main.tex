\begin{frame}[t]{Minimization with Newton's method}
	\vspace{-1em}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\vspace{-2em}
			\begin{tcolorbox}[colframe=gray,%
					sharp corners=all, left=0mm, bottom=0mm, leftright skip=-1mm]
				\textbf{Objective function minimization:}
				$$\vec{P}, \vec{Q} = \operatorname*{argmin}_{\vec{p}, \vec{q}} f(\vec{p}, \vec{q}, z, t_\text{diff}, t_\text{sum})$$
				{
						\scriptsize
						\textit{where}:
						\begin{flalign*}
							\setlength{\belowdisplayskip}{0pt}
							\setlength{\abovedisplayskip}{0pt}
							\hspace{3mm} \vec{p} & : \text{Parameters of scintillators}                                                   & \\
							\vec{q}              & : \text{Parameters of all muon tracks }(\vec{q}^0, \ \vec{q}^1, \ \ldots, \ \vec{q}^N) &
						\end{flalign*}
					}
			\end{tcolorbox}
			\begin{alertblock}{Limitations of Newton's method}
				\begin{itemize}
					\item Evaluation of Hessian is very expensive. (Quasi Newton's method, e.g. BFGS)
					\item Expensive computation with large numbers of parameters.
				\end{itemize}
			\end{alertblock}
			Total number of parameters using N muon tracks:
			$$3600 + 6N \sim \mathcal{O}(N)$$
		\end{column}
		\begin{column}{0.45\textwidth}
			\textbf{Newton's method}
			\begin{figure}
				% https://suzyahyah.github.io/calculus/optimization/2018/04/06/Taylor-Series-Newtons-Method.html
				\includegraphics[width=0.8\textwidth]{DPG2026/NewtonsMethod}
			\end{figure}
			Iteration step:
			\begin{flalign*}
				 &                          & \overrightarrow{W}_{n+1} & = \overrightarrow{W}_{n} - \eta \cdot \left.  \mathcal{B}(\vec{w})^{-1} \nabla f(\vec{w}) \right|_{\vec{w} = \overrightarrow{W}_{n}} \hspace{-6em} &                         & \\
				 & \text{\scriptsize where} &                          &                                                                                                                                                    &                           \\
				 &                          & \vec{w}                  & = \left(\vec{p}, \ \vec{q}^0, \ \vec{q}^1, \ \ldots, \ \vec{q}^N \right)                                                                           &                         & \\
				 &                          & \eta                     & = 1                                                                                                                                                & \text{(Line search)}    & \\
				 &                          & \mathcal{B}(\vec{q})     & = \nabla^2 f(\vec{q})                                                                                                                              & \text{(Hessian matrix)} &
			\end{flalign*}
		\end{column}
	\end{columns}
\end{frame}




\begin{frame}[t]{Newton's method and Millepede algorithm}
	\vspace{-3em}
	\begin{columns}[t]
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colframe=gray, halign title=flush center, adjusted title=Objective function minimization,%
					sharp corners=all, left=0mm, bottom=0mm]
				$$\vec{P}, \vec{Q} = \operatorname*{argmin}_{\vec{p}, \vec{q}} f(\vec{p}, \vec{q}, z, t_\text{diff}, t_\text{sum})$$
				{
						\scriptsize
						\textit{where}:
						\begin{flalign*}
							\setlength{\belowdisplayskip}{0pt}
							\setlength{\abovedisplayskip}{0pt}
							\hspace{3mm} \vec{p} & : \text{Parameters of scintillators} & \\
							\vec{q}              & : \text{Parameters of muon tracks}   &
						\end{flalign*}
					}
			\end{tcolorbox}
			\textbf{Traditional Newton's method:}
			\vspace{-1em}
			\begin{figure}
				% https://suzyahyah.github.io/calculus/optimization/2018/04/06/Taylor-Series-Newtons-Method.html
				\includegraphics[width=0.6\textwidth]{DPG2026/NewtonsMethod}
			\end{figure}
			\vspace{-1em}
			{\scriptsize
			Iteration step:
			$$\overrightarrow{W}_{n+1}= \overrightarrow{W}_{n} - \eta \cdot \left.  \mathcal{B}(\vec{w})^{-1} \nabla f(\vec{w}) \right|_{\vec{w} = \overrightarrow{W}_{n}} $$
			\textit{where}:
			\begin{flalign*}
				\hspace{10em}	\eta   & = 1                   & \text{(Line search)}    & \\
				\mathcal{B}(\vec{q}) & = \nabla^2 f(\vec{q}) & \text{(Hessian matrix)} &
			\end{flalign*}
			}
			\vspace{-2em}
		\end{column}
		\begin{column}{0.55\textwidth}
			\begin{alertblock}{Limitations of Newton's method}
				\begin{itemize}
					\item Evaluation of Hessian is very expensive. (Quasi Newton's method, e.g. BFGS)
					\item Expensive computation with large numbers of parameters.
				\end{itemize}
			\end{alertblock}
			\textbf{Millepede algorithm:}

			Separation into global and local parameters:
			{\scriptsize
			$\begin{array} {|c|c|c|c|} \hline
					\text{Global parameters} \vec{p}_i & \text{Local paramters} \ \vec{q}^0 & ... & \text{Local parameters} \ \vec{q}^N \\ \hline
					p_1, p_2, \ldots, p_{3900}         & a^0_x, b^0_x, a^0_t, b^0_t         & ... & a^N_x, b^N_x, a^N_t, b^N_t          \\ \hline
				\end{array}$

			}
			\footnotesize Iteration on global parameters:
			$$ $$
			\begin{flalign*}
				 &                 & \vec{p}_{n+1}                  & = \vec{p}_{n} - \left.  \mathcal{B}(\vec{p}, \vec{q})^{-1} \left(\nabla_p f + \mathcal{R}(\vec{p}, \vec{q})^{-1}\nabla_q f \right) \right|_{\vec{p} = \vec{p}_{n}, \vec{q} = \vec{q}^n} & \\
				 & \textit{where}: &                                &                                                                                                                                                                                         & \\
				 &                 & \mathcal{B}(\vec{p}, \vec{q})  & =  \nabla^2_pf  -\nabla^2_{pq}f \cdot \nabla^2_{q}f \cdot \left(\nabla^2_{pq}f\right)^\intercal                                                                                         & \\
				 &                 & \mathcal{R} (\vec{p}, \vec{q}) & = - \nabla^2_{pq}f \cdot \left(\nabla^2_qf\right)^{-1}                                                                                                                                  &
			\end{flalign*}

		\end{column}
	\end{columns}
\end{frame}
